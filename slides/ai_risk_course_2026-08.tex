\section{Act I --- What is a risk forecast?}
\begin{frame}[plain]

\begin{frame}{A risk model targets a different object}
  \begin{itemize}
    \item A \textbf{forecasting} model targets the centre of tomorrow's distribution;
          a \textbf{risk} model targets a quantile of it.
      \[ \hat r_{t} = \E\!\left[r_{t}\mid \mathcal{I}_{t-1}\right],
         \qquad
         \Prob\!\left(r_{t} < q_{t}(\alpha)\mid \mathcal{I}_{t-1}\right) = \alpha \]
    \item With $L_t = -r_t$ the loss and $q_{t}(\alpha) < 0$ the return quantile, Value
          at Risk and Expected Shortfall (ES) are:
      \[ \mathrm{VaR}_{t}(\alpha) = -q_{t}(\alpha) > 0, \qquad
         \mathrm{ES}_{t}(\alpha) = \frac{1}{\alpha}\int_{0}^{\alpha}
         \mathrm{VaR}_{t}(u)\,\mathrm{d}u \]
    \item VaR is a \textbf{threshold}; ES is an \textbf{average of the tail}.
      \begin{itemize}
        \item VaR locates the edge of the tail, ES averages what lies past it.
        \item On a continuous law that average is $\E[L_t \mid L_t > \mathrm{VaR}_t]$.
      \end{itemize}
    \item We score \textbf{VaR} today: every model family here can be asked for it.
      \begin{itemize}
        \item ES returns in the appendix, where scoring it needs the pair.
      \end{itemize}
  \end{itemize}
  \begin{itemize}
    \item[\textcolor{MainBlue}{$\boxdot$}] \textcolor{MainBlue}{Throughout: VaR and ES
          are \textbf{positive magnitudes}.}

  \end{itemize}
\end{frame}

\begin{frame}{Where do VaR and ES actually sit?}
	\centering
	\includegraphics[width=0.698\textwidth]{s_vares}
	\begin{block}{Two objects, and only one of them is a threshold}
		\begin{itemize}
				\item The \textbf{edge} of the shaded band is the VaR; the \textbf{mean}
				inside it is the ES.
				\item Here that edge sits at $-3.45$ and the tail mean at $-5.01$, so
				VaR $= 3.45$ and ES $= 5.01$.
				\item Right: a Gaussian and a Student-$t_{3}$, each scaled to that same VaR.
				\item Their expected shortfalls are $3.95$ and $5.32$, so VaR alone reports the
				same risk for both.
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}{What does coherence ask of a risk measure?}
  \begin{itemize}
    \item $\rho(L)$ is the \textbf{capital} a book with loss $L$ must hold.
      \begin{itemize}
        \item A measure with all four properties below is \textbf{coherent}.
      \end{itemize}
    \item \textbf{Monotonicity.} $L_{1} \le L_{2}$ always
          $\Rightarrow \rho(L_{1}) \le \rho(L_{2})$.
      \begin{itemize}
        \item More loss, more capital.
      \end{itemize}
    \item \textbf{Translation.} $\rho(L + c) = \rho(L) + c$ for a certain loss $c$.
      \begin{itemize}
        \item Cash set aside counts once, at face value.
      \end{itemize}
    \item \textbf{Homogeneity.} $\rho(\lambda L) = \lambda\,\rho(L)$ for
          $\lambda \ge 0$.
      \begin{itemize}
        \item Double the book, double the requirement.
      \end{itemize}
    \item \textbf{Sub-additivity.} $\rho(L_{1} + L_{2}) \le \rho(L_{1}) + \rho(L_{2})$.
      \begin{itemize}
        \item Diversification can only lower the requirement, and this is the one
              VaR loses.
      \end{itemize}
  \end{itemize}
  \vspace{0.3em}
  \begin{block}{}
    \begin{itemize}
      \item VaR satisfies the first three \parencite{artzner1999coherent}, and is
            sub-additive on \textbf{elliptical} losses: jointly Gaussian, or
            Student-$t$ sharing one degrees-of-freedom $\nu$.
      \item Outside it VaR can fail, and Basel moved capital to $97.5\%$ ES
            \parencite{bcbs2019frtb}.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Two bonds, and the axiom VaR loses}
  \centering
  \includegraphics[width=0.795\textwidth]{s_coherent}
  \begin{block}{Merging two books raised the requirement}
    \begin{itemize}
      \item Held alone, each default sits \emph{beyond} the $1\%$ quantile: VaR reports
            $\mathbf{0}$ on both books.
      \item Merged, $P(\text{at least one}) = \mathbf{1.59\%}$ crosses $\alpha$, so VaR
            becomes $\mathbf{100}$ --- while ES gives $80 + 80$ against
            $\mathbf{100.64}$ and holds.
    \end{itemize}
  \end{block}
\end{frame}



\begin{frame}{Two factors, but only one is richly identified}
  \begin{itemize}
    \item $\mu_{t}$ is the conditional mean, $\sigma_{t}$ the \textbf{scale}, and
          $q_{z}(\alpha)$ the quantile of the standardised shock $z_{t}$.
      \[ q_{t}(\alpha) = \mu_{t} + \scale{\sigma_{t}}\,\shape{q_{z}(\alpha)}
         \qquad\text{when}\qquad
         r_{t} = \mu_{t} + \sigma_{t}\,z_{t},\ \ z_{t}\ \text{i.i.d.} \]
      \vspace{-1.1em}
      \begin{itemize}
        \item A \emph{location--scale} law: the shape is the same on every day. Every
              classical benchmark assumes it.
      \end{itemize}
  \end{itemize}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{The scale}
        \begin{itemize}
          \item Re-estimated every day from the recent past.
          \item Volatility clusters, so it is forecastable.
          \item Thousands of examples to learn from.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{The shape}
        \begin{itemize}
          \item One observation per hundred days.
          \item A year of data holds \textbf{two or three}.
          \item The tail shape rests on those few.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{itemize}
    \item $1\%$ is not $5\%$ with a smaller number.
      \begin{itemize}
        \item It is the same question asked where the data runs out, and each method
              degrades differently on the way.
      \end{itemize}
    \item[\textcolor{IDAred}{$\boxdot$}] \textcolor{IDAred}{Every classical result
          today says which of the two it got wrong.}
  \end{itemize}
\end{frame}

\begin{frame}{Four questions people ask of one table}
  \begin{itemize}
    \item \textbf{\textcolor{MainBlue}{PIPELINE AUDIT}} --- did the object you scored
          come back from the question you asked?
      \begin{itemize}
        \item Prior to the other three, and the one Act III is about.
      \end{itemize}
    \item \textbf{\textcolor{MainBlue}{RANKING}} --- which forecast has the lower
          consistent loss?
      \begin{itemize}
        \item A relative statement: it orders two forecasts and stops there.
      \end{itemize}
    \item \textbf{\textcolor{MainBlue}{VALIDITY}} --- is the forecast compatible with
          the target coverage, and with the dynamics of the breaches?
      \begin{itemize}
        \item An absolute test against a null, one model at a time.
      \end{itemize}
    \item \textbf{\textcolor{IDAred}{POWER}} --- would that test have detected a
          materially invalid model on this sample?
      \begin{itemize}
        \item A property of the experiment, fixed before any model runs.
        \item Reported far less often than the other three.
      \end{itemize}
  \end{itemize}

    \begin{itemize}

    \end{itemize}

\begin{frame}{The simplest answer: sort the window}
  \begin{itemize}
    \item Historical simulation: sort the last $W$ days, read off the empirical
          quantile.
      \[ \widehat{\mathrm{VaR}}^{\mathrm{HS}}_{t}(\alpha)
         = -\,\widehat{Q}_{\alpha}\!\left(r_{t-W},\dots,r_{t-1}\right) \]
      \vspace{-1.1em}
      \begin{itemize}
        \item The window supplies the tail shape directly.
      \end{itemize}
    \item So the only modelling decisions left are $W$ and the weighting.
      \begin{itemize}
        \item Both are visible choices, and both move the answer.
      \end{itemize}
    \item Every day in the window carries weight $1/W$.
      \begin{itemize}
        \item At $W = 1000$, a day from four years ago counts as much as yesterday.
      \end{itemize}
    \item It reacts only \textbf{discretely}, when a day enters or leaves the window.
      \begin{itemize}
        \item So the response is coarse, and where it steps depends on $W$.
      \end{itemize}
  \end{itemize}
  \quantlet{AIDA\_aug\_data}{\qlbase/quantlets/AIDA_aug_data}
\end{frame}

\begin{frame}{GARCH(1,1): model the scale, assume the shape}
  \begin{itemize}
    \item One lag of the squared shock, one of the variance.
      \[ \varepsilon_{t} = r_{t} - \mu, \qquad
         \sigma_{t}^{2} = \omega + \alpha_{1}\varepsilon_{t-1}^{2}
                         + \beta_{1}\sigma_{t-1}^{2}, \qquad
         \widehat{\mathrm{VaR}}_{t}(\alpha)
            = -\left(\hat\mu + \hat\sigma_{t}\, q_{\hat\nu}(\alpha)\right) \]
      \vspace{-1.1em}
      \begin{itemize}
        \item $\mu$ is the constant mean and $\omega$ the variance intercept;
              $\alpha_{1}$ weights yesterday's shock, $\beta_{1}$ yesterday's variance.
        \item Subscripted, $\alpha_{1}$ is a GARCH coefficient; bare, $\alpha$ stays the
              VaR level throughout.
      \end{itemize}
    \item The tail \textbf{family is imposed} on the standardised shock
          $z_{t} = \varepsilon_{t}/\sigma_{t}$; only its parameters are estimated.
      \begin{itemize}
        \item Here it is \textbf{Student-$t$}.
        \item With $\nu$ the degrees of freedom, a raw $t_{\nu}$ has variance
              $\nu/(\nu-2)$, so it is rescaled to make $\mathrm{Var}(z_t) = 1$:
          \[ q_{\nu}(\alpha) = t_{\nu}^{-1}(\alpha)\,
             \sqrt{\tfrac{\nu-2}{\nu}} \]
          \vspace{-1.2em}
        \item At $\nu = 6.5$: $-3.06 \times 0.83 = \mathbf{-2.55}$. 
        \item Skip the rescaling
              and VaR comes out \textbf{20\%} too large.
      \end{itemize}
    \item This is the scale/shape split written as one equation.
      \begin{itemize}
        \item Published in 1982 and 1986 \parencite{engle1982arch,bollerslev1986garch},
              and still the benchmark to beat.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{What the squared shock costs you}
  \centering
  \includegraphics[width=0.776\textwidth]{s_garch}
  \points{%
      \item Shocks of $-10\%$ and $+10\%$ leave the same $\sigma_t$, because the shock enters squared.
      \item Volatility usually reacts more to a fall than to an equal rise --- the \textbf{leverage effect}, which sits outside this model.
      \item Right: the scale moves every day, where historical simulation holds still and steps.
  }
\end{frame}

\begin{frame}{Quantile regression: what goes in}
  \begin{itemize}
    \item Fitted on \textbf{realised} pairs from the estimation window: 
    \begin{itemize}
    	\item the return  $r_{t}$, against predictors $x_{t-1}$.
      \end{itemize}
    \item Seven columns:
        \begin{itemize}
        	\item  $r_{t-1}$, $|r_{t-1}|$, the $22$-day minimum, three
          \textbf{realised volatilities} and one \textbf{EWMA}.
      \[ \mathrm{RV}^{(w)}_{t} = \sqrt{\tfrac{1}{w}\textstyle\sum_{i=1}^{w}
         r_{t-i}^{2}}\ \ (w = 5, 22, 66), \qquad
         \sigma^{2}_{t} = \lambda\sigma^{2}_{t-1} + (1-\lambda)r^{2}_{t-1},\
         \lambda = 0.94 \]
         
      \vspace{1em}

        \item All estimates of the same \scale{scale}, equally against exponentially
              weighted.

      \end{itemize}
    \item Choose a function $f(x_{t-1};\theta)$ mapping those predictors to one number.
      \begin{itemize}
        \item Linear $f = \theta^{\top}x_{t-1}$, a tree ensemble, or a network.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Quantile regression (QR): the loss decides what you fit}
  \begin{itemize}
    \item Fit $\theta$ with the \textbf{pinball} loss
          \parencite{koenker1978quantiles} in place of squared error.
      \[ L_{\alpha}(r, q) = \left(\alpha - \mathbf{1}\{r < q\}\right)(r - q),
         \qquad
         \hat\theta = \arg\min_{\theta}\sum_{t}
         L_{\alpha}\!\left(r_{t}, f(x_{t-1};\theta)\right) \]
      \vspace{-1.1em}
      \begin{itemize}
        \item Squared error would place $f$ at the conditional \emph{mean}.
        \item QR places it at the conditional \emph{quantile}.
        \item To forecast, feed it \emph{today's} predictors:
              $\widehat{\mathrm{VaR}}_{t+1}(\alpha) = -f(x_{t};\hat\theta)$.
      \end{itemize}
    \item The loss is \textbf{strictly consistent} for the quantile.
      \begin{itemize}
        \item Minimised at the true quantile, so every ranking here uses it
              \parencite{gneiting2011point}.

      \end{itemize}
    \item The difficulty is the \emph{level} you point it at.
      \begin{itemize}
        \item Near $1\%$ the surface is locally flat.
        \item So $f$ fits the centre while the \shape{tail} drifts.
      \end{itemize}
  \end{itemize}
\end{frame}

  \points{%

      \item Squared error puts the line at the mean: slope $\mathbf{+0.01}$, flat and useless for risk.
      \item Pinball at $1\%$ puts it at the quantile: slope $\mathbf{-1.94}$, steep in volatility.

\begin{frame}{Where can AI actually help?}
  \begin{itemize}
    \item \scale{\textbf{The scale}} has thousands of training examples.
      \begin{itemize}
        \item Every day is one, and volatility persists: an ordinary regression.
      \end{itemize}
    \item \shape{\textbf{The shape}} has a handful.
      \begin{itemize}
        \item Two or three tail events a year, whatever the model.
        \item The count belongs to the data, and every architecture faces it.
      \end{itemize}
    \item So hand the learner the scale, and keep the shape parametric.
      \begin{itemize}
        \item \textbf{NN-$t$} --- a \textbf{neural network} for the scale, a
              Student-$t$ for the tail. It predicts $\log\sigma_{t}$ from the same
              seven predictors, and is scored in every table from here on.
          \[ \sigma_{t} = \exp g(x_{t-1};\theta), \qquad
             \hat\theta = \arg\max_{\theta}\sum_{t}
             \left[\log f_{\nu}\!\left(\tfrac{r_{t}}{\sigma_{t}}\right)
             - \log\sigma_{t}\right], \qquad
             \widehat{\mathrm{VaR}}_{t}(\alpha)
             = -\hat\sigma_{t}\,q_{\hat\nu}(\alpha) \]
          \vspace{-1.1em}
        \item GARCH's own likelihood, with the variance \emph{learned} rather than
              imposed by a recursion.
      \end{itemize}
  \end{itemize}
  \begin{block}{}
    \begin{itemize}
      \item A pretrained model claims \emph{both} factors at once, from a corpus of
            other series.
        \begin{itemize}
          \item Any gain must come from transferable structure, extra information or an
                assumption, and each of the three is checkable.
        \end{itemize}
    \end{itemize}
  \end{block}
\end{frame}

  }

\begin{frame}{Chronos-T5 forecasts the next value, so we hand it prices}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item Chronos-T5 was trained on series \textbf{levels}, and rescales each
              context by its own mean and spread.
          \begin{itemize}
            \item So it gets a \emph{context} of the last $512$ closing prices, the
                  length it was trained to read, and returns a distribution for the next.
          \end{itemize}
        \item That distribution is over prices, and we need one over returns.
          \begin{itemize}
            \item Each sampled price $\hat P_{t}$ becomes a return against the last
                  observed close $P_{t-1}$:
                  $\hat r_{t} = 100\log(\hat P_{t}/P_{t-1})$.
          \end{itemize}
        \item That step preserves the quantile, because the map is increasing in
              $\hat P_{t}$.
          \begin{itemize}
            \item The $\alpha$ quantile of the price is therefore the $\alpha$ quantile
                  of the return.
          \end{itemize}
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{alertblock}{What happens if you feed returns instead}
        \begin{itemize}
          \item A return series is already near zero mean with a small spread, so the
                rescaling has nothing left to do.
            \begin{itemize}
              \item The predictive spread collapses, and the VaR collapses with it.
            \end{itemize}
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Five different things people mean by ``LLMs for risk''}
  \begin{enumerate}
    \item \textbf{Feature extraction} --- sentiment from text, fed to a conventional
          model.
      \begin{itemize}
        \item The AI stops at a feature. The risk number is still econometrics.
      \end{itemize}
    \item \textbf{Event classification} --- label the news flow.

        \item Still a feature: the risk model consumes the label.
      \end{itemize}
    \item \textbf{Probability elicitation} --- ask for $\Prob(\text{tail event})$.

    \item \textbf{Predictive simulation} --- sample tomorrow's \emph{return}, many times.
      \begin{itemize}
        \item The draws are a distribution; the quantile is computed from it.
      \end{itemize}
  \end{enumerate}

\begin{frame}{Backtesting in one line}
  \hypertarget{main:backtest}{}%
  \begin{itemize}
    \item Count the days the threshold was crossed, over $T$ scored days:
      \[ \hat\alpha = \frac{1}{T}\sum_{t=1}^{T}
         \mathbf{1}\!\left\{r_{t} < -\widehat{\mathrm{VaR}}_{t}(\alpha)\right\} \]
  \end{itemize}
  \begin{itemize}
    \item \textbf{Kupiec} tests whether $\hat\alpha$ differs from $\alpha$
          \parencite{kupiec1995}.
      \begin{itemize}
        \item $LR_{UC}\sim\chi^{2}_{1}$ under the null.
      \end{itemize}
    \item \textbf{Christoffersen} \parencite{christoffersen1998} tests whether the
          breaches \emph{cluster}.
      \begin{itemize}
        \item Right count, wrong timing is still a failure: breaches that arrive in one
              fortnight are the fortnight that matters.
      \end{itemize}
    \item Their sum is conditional coverage, $\chi^{2}_{2}$.
      \begin{itemize}
        \item The full expressions, and the Diebold--Mariano (DM) test that compares two
              models, are in the appendix.
      \end{itemize}
  \end{itemize}
  \vspace{0.3em}
  \centering

  \hyperlink{ap:dm}{\beamergotobutton{Diebold--Mariano}}\;

\begin{frame}{One page of conventions this course assumes}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Signs and objects}
        \begin{itemize}
          \item $r_t$ is the log return in \%; the loss is $L_t = -r_t$.
          \item $q_t(\alpha) = F_t^{-1}(\alpha) < 0$ is the return quantile.
          \item $\mathrm{VaR}_t(\alpha) = -q_t(\alpha) > 0$: a positive magnitude.
          \item $\mathrm{ES}_t(\alpha) = \frac{1}{\alpha}\int_{0}^{\alpha}\mathrm{VaR}_t(u)\,\mathrm{d}u$:
                the worst $\alpha$ averaged.
          \item Breach indicator: $\mathbf{1}\{r_t < -\widehat{\mathrm{VaR}}_t(\alpha)\}$.
          \item A \textbf{hat} marks a quantity computed from data.
        \end{itemize}
      \end{block}
      \begin{block}{Scale and shape}
        \begin{itemize}
          \item $q_t(\alpha) = \mu_t + \sigma_t\, q_z(\alpha)$ under a
                location--scale law: scale re-estimated daily, shape held fixed.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{The pinball loss}
        \begin{itemize}
          \item $L_\alpha(r, q) = (\alpha - \mathbf{1}\{r < q\})(r - q)$.
          \item Minimised at the true quantile, so it ranks forecasts.
          \item At $\alpha = 1\%$ the arms have slopes $0.01$ and $-0.99$.
        \end{itemize}
      \end{block}
      \begin{alertblock}{How many breaches to expect at $1\%$}
        \begin{itemize}
          \item 250 days: $2.5$ expected; 500 days: $5$.
          \item Which is why Act V spends time on power.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What makes a loss the right loss?}
  \hypertarget{ap:loss}{}%
  \hfill\hyperlink{main:backtest}{\beamerreturnbutton{back}}\par\vspace{-0.6em}
  \begin{defn}[Consistency]
    A scoring function $L$ is \textbf{consistent} for a functional $T$ if
    $\E\left[L(Y, T(F))\right] \le \E\left[L(Y, x)\right]$ for every $F$ and every
    report $x$. It is \textbf{strictly} consistent if equality forces $x = T(F)$.
  \end{defn}
  \vspace{0.3em}
  \begin{defn}[Elicitability]
    $T$ is \textbf{elicitable} if some strictly consistent $L$ exists for it. Then
    $T(F) = \arg\min_{x} \E\left[L(Y, x)\right]$, and models can be ranked by average
    loss.
  \end{defn}
  \vspace{0.4em}
  \begin{itemize}
    \item Quantiles are elicitable; the pinball loss does it \parencite{gneiting2011point}.
    \item ES \emph{lacks} one of its own, which Act V takes up.
  \end{itemize}
\end{frame}

\begin{frame}{Kupiec and Christoffersen in full}
  \hypertarget{ap:tests}{}%
  \hfill\hyperlink{main:backtest}{\beamerreturnbutton{back}}\par\vspace{-0.6em}
  \begin{itemize}
    \item With $T$ days, $x$ breaches and $n_{ij}$ transitions between breach and
          no-breach:
  \end{itemize}
  \begin{defn}[Unconditional coverage]
    \vspace{-0.6em}
    \[ LR_{UC} = -2\log\frac{(1-\alpha)^{T-x}\,\alpha^{x}}
       {\left(1-\tfrac{x}{T}\right)^{T-x}\left(\tfrac{x}{T}\right)^{x}}
       \ \sim\ \chi^{2}_{1} \]
    \vspace{-0.4em}
    \hfill{\scriptsize\parencite{kupiec1995}}
  \end{defn}
  \begin{defn}[Independence]
    \vspace{-0.6em}
    \[ LR_{IND} = -2\log\frac{(1-\hat\pi)^{n_{00}+n_{10}}\hat\pi^{\,n_{01}+n_{11}}}
       {(1-\hat\pi_{01})^{n_{00}}\hat\pi_{01}^{\,n_{01}}
        (1-\hat\pi_{11})^{n_{10}}\hat\pi_{11}^{\,n_{11}}}
       \ \sim\ \chi^{2}_{1} \]
    \vspace{-0.4em}
    \hfill{\scriptsize\parencite{christoffersen1998}}
  \end{defn}
  \vspace{0.2em}
  \begin{itemize}
    \item $\hat\pi_{i1} = n_{i1}/(n_{i0}+n_{i1})$ is the breach probability after state
          $i$; $\hat\pi$ pools them.
      \begin{itemize}
        \item Conditional coverage is the sum of the two, $\chi^{2}_{2}$.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Comparing two models on the same days}
  \hypertarget{ap:dm}{}%
  \hfill\hyperlink{main:backtest}{\beamerreturnbutton{back}}\par\vspace{-0.6em}
  \begin{defn}[Diebold--Mariano]
    With $d_{t} = L_{\alpha}(r_{t}, q^{A}_{t}) - L_{\alpha}(r_{t}, q^{B}_{t})$,
    \vspace{-0.4em}
    \[ DM = \frac{\bar d}{\sqrt{\hat\Omega / T}} \ \xrightarrow{d}\ \mathcal{N}(0,1),
       \qquad H_{0}:\ \E[d_{t}] = 0 \]
    \vspace{-0.4em}
    \hfill{\scriptsize\parencite{diebold1995comparing}}
  \end{defn}
  \begin{itemize}
    \item $\hat\Omega$ is a \textbf{HAC} estimate \parencite{newey1987hac}.
      \begin{itemize}
        \item Loss differentials are serially dependent, so the naive variance can
              overstate significance.
      \end{itemize}
    \item The test is paired, so both models must be scored on the \emph{same} days.
      \begin{itemize}
        \item A model that skips days is not comparable until the sample is aligned.
      \end{itemize}
    \item Report $\bar d$ and its interval alongside the verdict.
  \end{itemize}
\end{frame}

\begin{frame}{Scoring ES means scoring the pair}
  \begin{defn}[Fissler--Ziegel FZ$_{0}$ loss]
    For a quantile $v$ and a shortfall $e$ with $e < v < 0$ at level $\alpha$,
    \vspace{-0.4em}
    \[ L_{\mathrm{FZ}_{0}}(r, v, e) =
       -\frac{1}{\alpha e}\mathbf{1}\{r \le v\}(v - r)
       + \frac{v}{e} + \log(-e) - 1 \]
    \vspace{-0.4em}
    \hfill{\scriptsize\parencite{fissler2016elicitability,patton2019dynamic}}
  \end{defn}
  \begin{itemize}
    \item ES \textbf{lacks a scoring function of its own}.
      \begin{itemize}
        \item Every candidate loss is minimised somewhere else, so ranking on ES alone
              stays out of reach.
      \end{itemize}
    \item The \emph{pair} $(\mathrm{VaR}, \mathrm{ES})$ is jointly elicitable.
      \begin{itemize}
        \item Which is why regulation can still be backtested
              \parencite{nolde2017elicitability}.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{SYNCRISK: when every desk runs the same model}
  \[ D_{t} = \max_{i}\widehat{\mathrm{VaR}}_{i,t} - \min_{i}\widehat{\mathrm{VaR}}_{i,t},
     \qquad
     \rho_{ij} = \mathrm{corr}\!\left(\widehat{\mathrm{VaR}}_{i},
     \widehat{\mathrm{VaR}}_{j}\right) \]
  \begin{itemize}
    \item If every desk runs a similar model, every desk de-risks the same morning.
    \item Shared \textbf{pretraining} is a new channel for that.
      \begin{itemize}
        \item The same weights behind many vendors, invisible from outside.
      \end{itemize}
    \item Measured on \textbf{changes}: $\lambda_1/N = 0.20$ against $0.11$.
      \begin{itemize}
        \item Above independence, and far below the $0.53$ a level correlation reports.
              The forecasts are persistent; correlating levels reads trend as agreement.
      \end{itemize}
    \item Measuring $\rho_{ij}$ across your own model set is a supervisory question.
      \begin{itemize}
        \item You compute both quantities this afternoon.
      \end{itemize}
  \end{itemize}
\end{frame}
