\begin{frame}{Four ways to produce one number}
  \begin{itemize}
    \item Tomorrow's $1\%$ value-at-risk, from four kinds of model.
  \end{itemize}
  \begin{itemize}
    \item A \textbf{statistical benchmark} you can compute by hand.
      \begin{itemize}
        \item Sort the window, read off the quantile. No parameters at all.
      \end{itemize}
    \item A \textbf{time-series foundation model}, pretrained elsewhere.
      \begin{itemize}
        \item It has never seen this asset, and you cannot inspect what it has seen.

  \end{itemize}
  \vspace{0.4em}

    \begin{itemize}
      \item It is whether you can tell, \emph{from the output alone}, that the model

    \end{itemize}
  \end{alertblock}

\begin{frame}[plain]
  \centering\vfill
  {\Large\textcolor{IDAred}{Act I}}\\[0.4em]
  {\large What is a risk forecast?}
  \vfill
\end{frame}

\begin{frame}{One targets the middle, the other the tail}
  \begin{itemize}
    \item A \textbf{forecasting} model targets the centre of tomorrow's distribution.
      \[ \hat r_{t} = \E\!\left[r_{t}\mid \mathcal{I}_{t-1}\right] \]
    \item A \textbf{risk} model targets a quantile of it.
      \[ \widehat{\mathrm{VaR}}_{t}(\alpha)\ \text{such that}\quad
         \Prob\!\left(r_{t} < -\widehat{\mathrm{VaR}}_{t}(\alpha)\mid
         \mathcal{I}_{t-1}\right) = \alpha \]
  \end{itemize}
  \vspace{0.3em}
  \begin{block}{}
    \begin{itemize}
      \item An excellent $R^{2}$ on the centre says nothing about the $1\%$ quantile.
        \begin{itemize}
          \item They are different summaries of the same conditional law.
        \end{itemize}
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{VaR asks how far; ES asks how bad beyond that}
  \begin{itemize}
    \item With $L_t = -r_t$ the loss and $q_{t}(\alpha) = F_{t}^{-1}(\alpha) < 0$ the
          return quantile:
      \[ \mathrm{VaR}_{t}(\alpha) = -q_{t}(\alpha) > 0, \qquad
         \mathrm{ES}_{t}(\alpha) = \E\!\left[L_{t}\mid L_{t} >
         \mathrm{VaR}_{t}(\alpha)\right] \]
  \end{itemize}
  \vspace{-0.3em}
  \begin{itemize}
    \item Convention for the whole deck: VaR and ES are \textbf{positive magnitudes}.
      \begin{itemize}
        \item Every figure is drawn in \textbf{return space}, where the same object is
              the negative threshold $q_{t}(\alpha)$.
      \end{itemize}
  \end{itemize}
  \begin{itemize}
    \item VaR is a \textbf{threshold}.
      \begin{itemize}
        \item It says nothing about what lies past it, and it is not sub-additive.
      \end{itemize}
    \item ES is an \textbf{average of the tail}.
      \begin{itemize}
        \item It is coherent \parencite{artzner1999coherent,acerbi2002coherence}, and Basel
              moved market-risk capital from $99\%$ VaR to $97.5\%$ ES for that
              reason \parencite{bcbs2019frtb}.
      \end{itemize}
  \end{itemize}
  \vspace{0.4em}
  \begin{itemize}
    \item We score VaR today.
      \begin{itemize}
        \item It is the one object every model family here can be asked for directly.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Point forecast, or the whole predictive distribution?}
  \centering
  \includegraphics[width=0.82\textwidth]{fig_02_point_vs_distribution}
  \takeaway{A risk model must deliver a distribution, or at least one quantile of it.
  Point accuracy is not the currency.}
\end{frame}

\begin{frame}{Two factors, and only one of them has data}
  \[ q_{t}(\alpha) = \scale{\sigma_{t}} \times \shape{q_{\nu}(\alpha)},
     \qquad
     \mathrm{VaR}_{t}(\alpha) = -\,q_{t}(\alpha) > 0 \]
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{The scale}
        \begin{itemize}
          \item One observation every day.
          \item Volatility clusters, so it is forecastable.
          \item A learning algorithm has data to learn from.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{The shape}
        \begin{itemize}
          \item At $\alpha = 1\%$: \textbf{five} events in 500 days.
          \item The tail index rests on a handful of observations.
          \item No architecture changes that.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.7em}
  \begin{itemize}
    \item[\textcolor{IDAred}{$\boxdot$}] \textcolor{IDAred}{Every result today is a
          statement about which of the two a given model got wrong.}
  \end{itemize}
\end{frame}

\begin{frame}{Why $1\%$ is a different problem}
  \begin{itemize}
    \item $1\%$ is the level supervision counts and the level that hurts.
      \begin{itemize}
        \item Two or three days a year, and they are the days that end careers.
      \end{itemize}
    \item It is also where every estimator is at its weakest.
      \begin{itemize}
        \item A 250-day historical-simulation window holds \textbf{two and a half}
              observations below the quantile.
        \item A 500-draw sampling model puts \textbf{five} draws below it.
        \item A 500-day backtest expects \textbf{five} breaches.
      \end{itemize}
    \item So $1\%$ is not $5\%$ with a smaller number.
      \begin{itemize}
        \item It is the same question asked where the data runs out, and every method
              here degrades differently as it gets there.
      \end{itemize}
  \end{itemize}
\end{frame}

  {\Large\textcolor{IDAred}{Act II}}\\[0.4em]
  {\large The classical answers}
  \vfill
\end{frame}

\begin{frame}{The simplest answer: sort the window}
  \begin{itemize}
    \item Historical simulation: sort the last $W$ days, read off the empirical
          quantile.
      \[ \widehat{\mathrm{VaR}}^{\mathrm{HS}}_{t}(\alpha)
         = -\,\widehat{Q}_{\alpha}\!\left(r_{t-W},\dots,r_{t-1}\right) \]
  \end{itemize}
  \begin{itemize}
    \item Nothing is fitted and nothing is assumed about the shape of the tail.
      \begin{itemize}
        \item There is no parametric distribution to mis-specify. Window length and
              weighting remain modelling decisions.
      \end{itemize}
    \item Every day in the window carries weight $1/W$.
      \begin{itemize}
        \item Including one from four years ago, at full weight.
      \end{itemize}
    \item It therefore \textbf{cannot react}.
      \begin{itemize}
        \item The threshold moves only when an extreme day enters or leaves the window.
      \end{itemize}
  \end{itemize}
  \quantlet{AIDA\_aug\_data}{\qlbase/quantlets/AIDA_aug_data}
\end{frame}

\begin{frame}{GARCH: model the scale, assume the shape}
  \vspace{-0.2em}
  \hfill{\scriptsize\textcolor{DarkGray}{\textcite{engle1982arch};
  \textcite{bollerslev1986garch}}}\vspace{-0.4em}
  \[ \sigma_{t}^{2} = \omega + \alpha_{1} r_{t-1}^{2} + \beta_{1}\sigma_{t-1}^{2},
     \qquad
     \widehat{\mathrm{VaR}}_{t}(\alpha) = -\left(\mu + \sigma_{t}\, q_{\nu}(\alpha)\right) \]
  \begin{itemize}
    \item The shock enters \textbf{squared}.
      \begin{itemize}
        \item A $-10\%$ day and a $+10\%$ day move $\sigma_t$ identically.
      \end{itemize}
    \item The tail \textbf{family is imposed}; its parameters are estimated.
      \begin{itemize}
        \item $q_{\nu}(\alpha)$ comes from the Student-$t$ assumption, with $\nu$ fitted.
        \item That is the scale/shape split made explicit in one equation.
      \end{itemize}
    \item Half a century old, and still the benchmark machine learning must beat.
  \end{itemize}
\end{frame}

\begin{frame}{Quantile regression: learn the quantile end to end}
  \begin{itemize}
    \item Minimise the \textbf{pinball} (check) loss
          \parencite{koenker1978quantiles} directly, with $q_t$ the forecast quantile:
      \[ L_{\alpha}(r_{t}, q_{t}) = \left(\alpha - \mathbf{1}\{r_{t} < q_{t}\}\right)
         \left(r_{t} - q_{t}\right) \]
  \end{itemize}
  \begin{itemize}
    \item It is \textbf{strictly consistent} for the quantile.
      \begin{itemize}
        \item Minimised in expectation at the truth, which makes it the right thing to
              rank on --- and the breach count is not \parencite{gneiting2011point}.
      \end{itemize}
    \item Any function class drops in: linear, boosted trees, a neural net.
    \item The catch is where we are standing.
      \begin{itemize}
        \item All observations enter the objective, but near $1\%$ identification is
              weak and the surface is locally flat.
        \item A flexible model fits the centre while the tail drifts.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{What makes a loss the right loss?}
  \begin{defn}[Consistency]
    A scoring function $L$ is \textbf{consistent} for a functional $T$ if
    $\E\left[L(Y, T(F))\right] \le \E\left[L(Y, x)\right]$ for every $F$ and every
    report $x$. It is \textbf{strictly} consistent if equality forces $x = T(F)$.
  \end{defn}
  \vspace{0.3em}
  \begin{defn}[Elicitability]
    $T$ is \textbf{elicitable} if some strictly consistent $L$ exists for it. Then
    $T(F) = \arg\min_{x} \E\left[L(Y, x)\right]$, and models can be ranked by average
    loss.
  \end{defn}
  \vspace{0.4em}
  \begin{itemize}
    \item Quantiles are elicitable; the pinball loss does it \parencite{gneiting2011point}.
    \item ES is \emph{not} elicitable on its own --- which is Act V's problem, not this
          one.
  \end{itemize}
\end{frame}

\begin{frame}{The pinball loss, and why $1\%$ is the hard case}
  \centering
  \includegraphics[width=0.93\textwidth]{s_pinball}
  \takeaway{Left: at $\alpha = 1\%$ the two arms have slopes $0.01$ and $-0.99$, so the loss
  punishes a threshold that is too close far harder than one too far. Right: the mean
  loss is minimised at the sample quantile, but at $1\%$ every candidate within
  \textbf{0.64} points of it is inside $1\%$ of the minimum --- against $0.46$ at $5\%$.
  Identification is weakest exactly where supervision looks.}
\end{frame}

\begin{frame}{Where can AI actually help?}
  \begin{itemize}
    \item \scale{\textbf{The scale.}} Thousands of training examples, a well-posed
          regression.
      \begin{itemize}
        \item Genuine gains. A small network on log-volatility beats EWMA.
      \end{itemize}
    \item \shape{\textbf{The shape.}} A handful of events.
      \begin{itemize}
        \item Flexibility here manufactures confidence, not information.
      \end{itemize}
    \item \textbf{The joint law.} What a pretrained model claims to deliver.
      \begin{itemize}
        \item Both factors at once, having never seen this asset. That is Act III.
      \end{itemize}
  \end{itemize}
  \vspace{0.7em}
  \begin{block}{}
    \begin{itemize}
      \item A flexible model does not create tail observations.
        \begin{itemize}
          \item It redistributes the ones you have.
        \end{itemize}
    \end{itemize}
  \end{block}
\end{frame}

  {\Large\textcolor{IDAred}{Act III}}\\[0.4em]
  {\large Time-series foundation models}
  \vfill
\end{frame}

\begin{frame}{Four words people use interchangeably}
  \begin{itemize}
    \item \textbf{Supervised forecasting} --- fit on \emph{this} series, predict it.
    \item \textbf{Pretraining} --- fit once, on a large corpus of \emph{other} series.
    \item \textbf{Fine-tuning} --- take a pretrained model, continue training on yours.
    \item \textbf{Zero-shot} --- take a pretrained model, run it, change nothing.
  \end{itemize}
  \vspace{0.8em}
  \begin{block}{Everything in this Act is zero-shot}
    \begin{itemize}
      \item No parameter has seen the S\&P 500 in a way you control.
      \item What it saw in pretraining, you cannot inspect. That asymmetry returns in
            Act V as a validation problem, not a curiosity.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Point forecast, marginal quantiles, or a predictive law?}
  \begin{itemize}
    \item Three different outputs, routinely reported as if interchangeable:
  \end{itemize}
  \begin{enumerate}
    \item a \textbf{point} forecast $\hat r_{t}$;
    \item a set of \textbf{marginal quantiles} at fixed levels;
    \item a full \textbf{predictive distribution} you can integrate.
  \end{enumerate}
  \vspace{0.6em}
  \begin{itemize}
    \item VaR needs (2) \emph{at your level}, or (3).
    \item ES needs (3), or enough of (2) to approximate the tail mean.
    \item A model that gives you (2) on \textbf{its} grid has not given you (2) on yours.
      \begin{itemize}
        \item Which sounds pedantic until the next slide.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Chronos: the series as a language}
  \begin{itemize}
    \item Values are scaled and quantised into a fixed vocabulary of tokens.
    \item A sequence model is pretrained on a large corpus of time series.
    \item Forecasting is next-token prediction, decoded back to the value scale.
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{Two heads, and the difference is the whole Act}
    \begin{itemize}
      \item \textbf{Chronos-T5} samples paths, so any quantile can be read off.
      \item \textbf{Chronos-Bolt} outputs a \emph{fixed grid} of quantiles, far faster.
    \end{itemize}
  \end{block}
  \vspace{0.2em}
  \centering\footnotesize \textcite{ansari2024chronos}. TimesFM
  \parencite{das2024timesfm} is the same idea with a different decoder.
  \quantlet{AIDA\_aug\_chronos}{\qlbase/quantlets/AIDA_aug_chronos}
\end{frame}

\begin{frame}{Why the input is prices}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item In the tested pipeline, price \textbf{levels} are the compatible input
              representation.
          \begin{itemize}
            \item Give it the last 512 closes, ask for the next.
          \end{itemize}
        \item Recover the implied return:
          \[ r = 100\log\!\left(\hat P_{t} / P_{t-1}\right) \]
        \item A monotone transform.
          \begin{itemize}
            \item So a price quantile maps to the return quantile of the same level.
          \end{itemize}
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{alertblock}{What happens if you feed returns}
        \begin{itemize}
          \item Returns are near-zero-mean noise.
            \begin{itemize}
              \item Predictive standard deviation \textbf{0.14} against a realised
                    \textbf{1.22}, and the VaR collapses with it.
            \end{itemize}
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \centering\textcolor{DarkGray}{Measured in the certified pipeline, S\&P 500,
  5541 out-of-sample days.}
\end{frame}

  \centering
  \includegraphics[width=0.93\textwidth]{s_clamp}
  \takeaway{Chronos-Bolt was trained on the grid $0.1, 0.2, \dots, 0.9$. Requests below $0.1$

  \quantlet{AIDA\_aug\_chronos}{\qlbase/quantlets/AIDA_aug_chronos}
\end{frame}

\begin{frame}{Why this is the important slide}
  \begin{itemize}
    \item A pipeline that requested \texttt{quantile\_levels=[0.01]} and took the number
          would have produced:
  \end{itemize}
  \begin{itemize}
    \item a complete set of results, a leaderboard position, and a plot;
    \item no exception, no failed assertion, no missing value;
    \item a breach rate around $10\%$ --- \textbf{ten times} what it claims.
  \end{itemize}
  \vspace{0.5em}
    \begin{alertblock}{And the natural reading would have been wrong}
    \begin{itemize}
      \item ``The foundation model is badly calibrated in the tail'' is a statement
            about the \emph{pipeline}, dressed as a finding about the model.
      \item Ask what the artefact \emph{can} answer, not only what you want.
      \item Then check the values, because the call succeeded.
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}{The sampling head has a floor of its own}
  \begin{itemize}
    \item Chronos-T5 draws $N$ price paths and reads the quantile off the sample.
  \end{itemize}
  \begin{itemize}
    \item At $\alpha = 1\%$ with $N = 500$: \textbf{five} draws fall below the quantile.
      \begin{itemize}
        \item The forecast is an order statistic of five points.
      \end{itemize}
    \item The library default is $N = 20$.
      \begin{itemize}
        \item Which cannot express a $1\%$ quantile at all: no draw lies below it.
      \end{itemize}
    \item So the sample count is a modelling choice.
      \begin{itemize}
        \item With the same standing as the window length in historical simulation,
              and reported about as often as never.
      \end{itemize}
  \end{itemize}
\end{frame}

  {\large Language models as risk forecasters}
  \vfill

\begin{frame}{Four different things people mean by ``LLMs for risk''}
  \begin{enumerate}
    \item \textbf{Feature extraction} --- sentiment from text, fed to a conventional
          model.
      \begin{itemize}
        \item The AI stops at a feature. The risk number is still econometrics.
      \end{itemize}
    \item \textbf{Event classification} --- label the news flow.
    \item \textbf{Probability elicitation} --- ask for $\Prob(\text{tail event})$.
    \item \textbf{Quantile elicitation} --- ask for the VaR itself.
  \end{enumerate}
  \vspace{0.7em}
  \begin{block}{}
    \begin{itemize}
      \item Only (3) and (4) put the language model \emph{inside} the risk number.
        \begin{itemize}
          \item That is what we test, following \textcite{pele2026llmvar}.
        \end{itemize}
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Two ways to get a quantile out of a language model}
  \begin{block}{Sampling --- the published method}
    \[ \text{serialise} \;\rightarrow\; \text{sample } N \text{ completions}
       \;\rightarrow\; \widehat{Q}_{\alpha}\!\left(\text{draws}\right) \]
    \vspace{-0.7em}
    \begin{itemize}
      \item Statistically analogous to Chronos-T5 at the sampling-and-quantile stage:
            the quantile is an order statistic of the draws, and $N$ sets the floor.
    \end{itemize}
  \end{block}
  \begin{block}{Elicitation --- what we also run here}

\begin{frame}{The prompt is a modelling decision}
  \begin{itemize}
    \item Four information sets, and the differences between them are the experiment:
  \end{itemize}
  \begin{itemize}
    \item \texttt{series} --- the last 60 returns, \textbf{anonymised}.
      \begin{itemize}
        \item No ticker, no dates, no price levels.
      \end{itemize}
    \item \texttt{series+state} --- the same, plus volatility, drawdown, the HS VaR.
    \item \texttt{dated} --- the same returns, plus the asset name and the date.
    \item \texttt{dated+news} --- the same, plus the real headlines available at $t-1$.
  \end{itemize}

  \begin{alertblock}{Why two of them hide the date}
    \begin{itemize}
      \item The model's pretraining covers these dates.
        \begin{itemize}
          \item ``S\&P 500, 5 August 2024'' invites recall, and recall is not available
                for tomorrow.
        \end{itemize}
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}{One rule decides whether the headlines mean anything}
  \begin{block}{The as-of rule}
    \begin{itemize}
      \item A headline may enter the prompt for date $t$ only if it was published at or
            before the \textbf{as-of cutoff} on $t-1$.
        \begin{itemize}
          \item That cutoff is the asset's own market close.
        \end{itemize}
    \end{itemize}
  \end{block}
  \vspace{0.4em}
  \begin{itemize}
    \item The cutoff is measured per exchange, not assumed.
      \begin{itemize}
        \item S\&P $20{:}00$, DAX $15{:}30$, Nikkei $06{:}00$, gold $17{:}30$,
              Bitcoin $24{:}00$ UTC.
        \item New York's close applied to the Nikkei would hand it fourteen hours of
              news published after Tokyo shut.
      \end{itemize}
    \item Consecutive cutoffs \textbf{tile} the calendar.
      \begin{itemize}
        \item No article falls in two windows and none falls in none. A fixed 24-hour
              window silently drops every evening and every weekend.
      \end{itemize}
  \end{itemize}
  \quantlet{AIDA\_aug\_news}{\qlbase/quantlets/AIDA_aug_news}
\end{frame}

\begin{frame}{What can go wrong, in increasing order of danger}
  \begin{itemize}
    \item \textbf{Unparseable output.} The model wrote prose. Loud, and cheap to fix.

\begin{frame}{Backtesting in one line}
  \begin{itemize}
    \item Count the days the threshold was crossed:
      \[ \hat\alpha = \frac{1}{T}\sum_{t=1}^{T}
         \mathbf{1}\!\left\{r_{t} < -\widehat{\mathrm{VaR}}_{t}(\alpha)\right\} \]
  \end{itemize}
  \begin{itemize}
    \item \textbf{Kupiec} tests whether $\hat\alpha$ differs from $\alpha$
          \parencite{kupiec1995}.
      \begin{itemize}
        \item $LR_{UC}\sim\chi^{2}_{1}$ under the null.
      \end{itemize}
    \item \textbf{Christoffersen} \parencite{christoffersen1998} tests whether the
          breaches \emph{cluster}.
      \begin{itemize}
        \item Right count, wrong timing is still a failure: breaches that arrive in one
              fortnight are the fortnight that matters.
      \end{itemize}
    \item Their sum is conditional coverage, $\chi^{2}_{2}$.
  \end{itemize}
  \quantlet{AIDA\_aug\_backtest}{\qlbase/quantlets/AIDA_aug_backtest}
\end{frame}

\begin{frame}{Kupiec and Christoffersen in full}
  \begin{itemize}
    \item With $T$ days, $x$ breaches and $n_{ij}$ transitions between breach and
          no-breach:
  \end{itemize}
  \begin{defn}[Unconditional coverage]
    \vspace{-0.6em}
    \[ LR_{UC} = -2\log\frac{(1-\alpha)^{T-x}\,\alpha^{x}}
       {\left(1-\tfrac{x}{T}\right)^{T-x}\left(\tfrac{x}{T}\right)^{x}}
       \ \sim\ \chi^{2}_{1} \]
    \vspace{-0.4em}
    \hfill{\scriptsize\parencite{kupiec1995}}
  \end{defn}
  \begin{defn}[Independence]
    \vspace{-0.6em}
    \[ LR_{IND} = -2\log\frac{(1-\hat\pi)^{n_{00}+n_{10}}\hat\pi^{\,n_{01}+n_{11}}}
       {(1-\hat\pi_{01})^{n_{00}}\hat\pi_{01}^{\,n_{01}}
        (1-\hat\pi_{11})^{n_{10}}\hat\pi_{11}^{\,n_{11}}}
       \ \sim\ \chi^{2}_{1} \]
    \vspace{-0.4em}
    \hfill{\scriptsize\parencite{christoffersen1998}}
  \end{defn}
  \vspace{0.2em}
  \begin{itemize}
    \item $\hat\pi_{i1} = n_{i1}/(n_{i0}+n_{i1})$ is the breach probability after state
          $i$; $\hat\pi$ pools them.
      \begin{itemize}
        \item Conditional coverage is the sum of the two, $\chi^{2}_{2}$.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Comparing two models on the same days}
  \begin{defn}[Diebold--Mariano]
    With $d_{t} = L_{\alpha}(r_{t}, q^{A}_{t}) - L_{\alpha}(r_{t}, q^{B}_{t})$,
    \vspace{-0.4em}
    \[ DM = \frac{\bar d}{\sqrt{\hat\Omega / T}} \ \xrightarrow{d}\ \mathcal{N}(0,1),
       \qquad H_{0}:\ \E[d_{t}] = 0 \]
    \vspace{-0.4em}
    \hfill{\scriptsize\parencite{diebold1995comparing}}
  \end{defn}
  \begin{itemize}
    \item $\hat\Omega$ is a \textbf{HAC} estimate \parencite{newey1987hac}.
      \begin{itemize}
        \item Loss differentials are serially dependent; the naive variance overstates
              significance.
      \end{itemize}
    \item The test is paired, so both models must be scored on the \emph{same} days.
      \begin{itemize}
        \item A model that skips days is not comparable until the sample is aligned.
      \end{itemize}
    \item Report $\bar d$ and its interval, not only the verdict.
  \end{itemize}
\end{frame}

\begin{frame}{Scoring ES means scoring the pair}
  \begin{defn}[FZ$_{0}$ loss]
    For a quantile $v < 0$ and a shortfall $e < 0$ at level $\alpha$,
    \vspace{-0.4em}
    \[ L_{\mathrm{FZ}_{0}}(r, v, e) =
       -\frac{1}{\alpha e}\mathbf{1}\{r \le v\}(v - r)
       + \frac{v}{e} + \log(-e) - 1 \]
    \vspace{-0.4em}
    \hfill{\scriptsize\parencite{fissler2016elicitability,patton2019dynamic}}
  \end{defn}
  \begin{itemize}
    \item ES is not elicitable on its own.
      \begin{itemize}
        \item No scoring function is minimised by the true ES alone, so ES cannot be
              ranked the way a quantile can.
      \end{itemize}
    \item The \emph{pair} $(\mathrm{VaR}, \mathrm{ES})$ is jointly elicitable.
      \begin{itemize}
        \item Which is why regulation can still be backtested
              \parencite{nolde2017elicitability}.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Leakage: the failure that flatters every model}
  \begin{itemize}
    \item \textbf{Window leakage} --- the forecast day inside its own estimation window.
    \item \textbf{Alignment leakage} --- a return series shifted one row against dates.
      \begin{itemize}
        \item Silent, and it inflates every skill measure at once.
      \end{itemize}
    \item \textbf{Selection leakage} --- choosing the architecture on the reported period.
    \item \textbf{Temporal contamination} --- the pretraining corpus contains the outcome.
      \begin{itemize}
        \item The first three you can test for. The fourth you cannot, because you
              cannot read the corpus.
      \end{itemize}
  \end{itemize}
  \vspace{0.4em}
  \begin{block}{}
    \begin{itemize}
      \item Anonymising the prompt is therefore a design choice.
        \begin{itemize}
          \item And the news alignment is audited rather than trusted.
        \end{itemize}
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{SYNCRISK: when every desk runs the same model}
  \[ D_{t} = \max_{i}\widehat{\mathrm{VaR}}_{i,t} - \min_{i}\widehat{\mathrm{VaR}}_{i,t},
     \qquad
     \rho_{ij} = \mathrm{corr}\!\left(\widehat{\mathrm{VaR}}_{i},
     \widehat{\mathrm{VaR}}_{j}\right) \]
  \begin{itemize}
    \item If every desk runs a similar model, every desk de-risks the same morning.
    \item Shared \textbf{pretraining} is a new channel for that.
      \begin{itemize}
        \item The same weights behind many vendors, and no way to tell from outside.
      \end{itemize}
    \item Four prompts are not four opinions: their median $\rho$ is $0.75$.
      \begin{itemize}
        \item Historical simulation is the least correlated with the rest, at $0.21$.
      \end{itemize}
    \item Measuring $\rho_{ij}$ across your own model set is a supervisory question.
      \begin{itemize}
        \item You compute both quantities this afternoon.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{What AI adds to the work of checking}
  \begin{itemize}
    \item \textbf{You cannot inspect the training data.}
      \begin{itemize}
        \item Contamination is invisible in principle, not merely in practice.
      \end{itemize}
    \item \textbf{The artefact has properties you did not choose.}
      \begin{itemize}
        \item A quantile grid, a context length, a tokenisation, a sample count.
      \end{itemize}
    \item \textbf{The endpoint can change under you.}
      \begin{itemize}
        \item Reproducibility needs a pinned version and a stored prompt --- or open
              weights, which is the other reason they are in this package.
      \end{itemize}
    \item \textbf{Plausible output is the default.} It will always answer.
  \end{itemize}

      \item None of this argues against using these models.
        \begin{itemize}
          \item It argues for validating them harder than the ones you fitted yourself.
        \end{itemize}
    \end{itemize}
  \end{alertblock}
\end{frame}
